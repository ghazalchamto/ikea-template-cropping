\chapter{Appendix A: Label Schema Example}
\label{app:schema}

\Cref{lst:schema_example} shows an excerpt of the \texttt{1C50-IDBDM} schema file,
illustrating the structure of a full label-type definition.

\begin{lstlisting}[language={},
  caption={Excerpt from label-rules/1C50-IDBDM.json.},
  label={lst:schema_example}]
{
  "label_type_id": "1C50-IDBDM",
  "label_type_name": "Card 50x35mm - Identity B with DataMatrix",
  "physical_dimensions_mm": { "width": 50, "height": 35 },
  "shape": "card_landscape",
  "required_fields": [
    "ikea_logo", "product_name", "article_number",
    "internal_item_number", "origin_text", "supplier_name",
    "supplier_address", "copyright_notice", "plant_identifier",
    "human_readable_date", "custom_identifier",
    "dimensions_metric", "dimensions_imperial",
    "weight_compact", "copy_count_inline"
  ],
  "optional_fields": [
    "compliance_marks", "age_rating", "quantity_multiplier"
  ],
  "datamatrix": {
    "symbol_size": "26x26",
    "x_dimension_options_mm": [0.4],
    "gs1_ais_present": ["AI_240", "AI_13"]
  },
  "validation_rules": [
    {
      "rule_id": "R001",
      "field": "article_number",
      "check": "regex_match",
      "pattern": "\\d{3}\\.\\d{3}\\.\\d{2}",
      "severity": "error"
    },
    {
      "rule_id": "R002",
      "field": "supplier_name",
      "check": "exact_match",
      "value": "IKEA of Sweden AB",
      "severity": "error"
    }
  ]
}
\end{lstlisting}
